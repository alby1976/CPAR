' Paste into the worklist worksheet code window.
' Merge with existing event procedures; do not create duplicates.
Option Explicit

Private mUpdatingColumns As Boolean

Private Sub Worksheet_Calculate()
    If mUpdatingColumns Then Exit Sub
    On Error GoTo CleanUp
    mUpdatingColumns = True
    UpdateWorklistColumns
CleanUp:
    mUpdatingColumns = False
End Sub

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim lo As ListObject, watched As Range
    Dim i As Long, oldEvents As Boolean
    Dim errorText As String

    oldEvents = Application.EnableEvents
    On Error GoTo Failed
    Set lo = Me.ListObjects("tblFinalWorklist")
    If lo.DataBodyRange Is Nothing Then Exit Sub

    ' Only changes in actively filtered columns require reapplication.
    For i = 1 To lo.ListColumns.Count
        If lo.AutoFilter.Filters(i).On Then
            If watched Is Nothing Then
                Set watched = lo.ListColumns(i).DataBodyRange
            Else
                Set watched = Application.Union(watched, _
                    lo.ListColumns(i).DataBodyRange)
            End If
        End If
    Next i
    If watched Is Nothing Then Exit Sub
    If Application.Intersect(Target, watched) Is Nothing Then Exit Sub

    Application.EnableEvents = False
    lo.AutoFilter.ApplyFilter
    UpdateWorklistColumns
CleanExit:
    Application.EnableEvents = oldEvents
    Exit Sub
Failed:
    errorText = Err.Description
    Application.EnableEvents = oldEvents
    MsgBox "The edit may already be saved, but the filter could not be updated: " & _
           errorText, vbExclamation
End Sub
